## 2. Acción en `HistoriaController` de `ProyectoSecundario`

La lista de historias se obtiene usando el endpoint existente:

Agregar estas acciones al controlador:
```csharp
using Microsoft.AspNetCore.Mvc;
using Newtonsoft.Json;
using ProyectoSecundario.Auxiliar;
using ProyectoSecundario.Enum;
using ProyectoSecundario.Models;

namespace ProyectoSecundario.Controllers
{
    public class HistoriaController : Controller
    {
        private string baseUrlHistorias =
            "http://localhost:5281/api/Historia";

        [HttpGet]
        public IActionResult Finales(int? idHistoria)
        {
            string? token = HttpContext.Session.GetString("token");
            string? rol = HttpContext.Session.GetString("Rol");

            if (string.IsNullOrEmpty(token))
            {
                return RedirectToAction("Login", "Auth");
            }

            if (rol != "Administrador")
            {
                return RedirectToAction(nameof(Index));
            }

            ReporteFinalesModel modelo = new ReporteFinalesModel
            {
                IdHistoriaSeleccionada = idHistoria
            };

            // Obtiene las historias que se mostrarán en el combo.
            HttpResponseMessage respuestaHistorias =
                ClienteHttpAuxiliar.EnviarSolicitud(
                    baseUrlHistorias,
                    VerbosHttp.GET,
                    token: token
                );

            if (!respuestaHistorias.IsSuccessStatusCode)
            {
                ViewBag.Error =
                    "No se pudo obtener la lista de historias.";

                return View(modelo);
            }

            modelo.Historias =
                JsonConvert.DeserializeObject<List<HistoriaCatalogoModel>>(
                    ClienteHttpAuxiliar.ObtenerBody(respuestaHistorias))
                ?? new List<HistoriaCatalogoModel>();

            // Si el administrador seleccionó una historia,
            // obtiene sus capítulos y finales.
            if (idHistoria.HasValue && idHistoria.Value > 0)
            {
                HttpResponseMessage respuestaHistoria =
                    ClienteHttpAuxiliar.EnviarSolicitud(
                        baseUrlHistorias + "/" + idHistoria.Value,
                        VerbosHttp.GET,
                        token: token
                    );

                if (!respuestaHistoria.IsSuccessStatusCode)
                {
                    ViewBag.Error =
                        "No se pudo obtener la historia seleccionada.";

                    return View(modelo);
                }

                HistoriaModel? historia =
                    JsonConvert.DeserializeObject<HistoriaModel>(
                        ClienteHttpAuxiliar.ObtenerBody(respuestaHistoria));

                if (historia == null)
                {
                    ViewBag.Error = "La historia no existe.";
                    return View(modelo);
                }

                modelo.TituloHistoria = historia.Titulo;

                List<CapituloModel> finales =
                    historia.Capitulos?
                        .Where(c => c.Tipo == 1)
                        .ToList()
                    ?? new List<CapituloModel>();

                int mayorCantidad = finales.Any()
                    ? finales.Max(f => f.CantLlegaron)
                    : 0;

                modelo.Finales = finales
                    .Select(final => new FinalFrecuenciaModel
                    {
                        IdFinal = final.Id,
                        Titulo = final.Titulo,
                        CantidadVecesAlcanzado = final.CantLlegaron,

                        // En caso de empate, se indican todos los
                        // finales con la cantidad máxima.
                        EsMasFrecuente =
                            mayorCantidad > 0 &&
                            final.CantLlegaron == mayorCantidad
                    })
                    .OrderByDescending(f => f.CantidadVecesAlcanzado)
                    .ThenBy(f => f.Titulo)
                    .ToList();
            }

            return View(modelo);
        }

        public IActionResult Index()
        {
            return View();
        }
    }
}
```

> Si el endpoint `GET /api/Historia` devuelve `HistoriaDTO` completos y no `HistoriaCatalogoDTO`, se puede deserializar igualmente a una lista de `HistoriaModel`, o crear un modelo específico para la lista. Lo importante es que existan `Id` y `Titulo`.

```csharp
[HttpGet("{id}")]
[Authorize(Roles = "Administrador")]

[ProducesResponseType((int)HttpStatusCode.BadRequest)]
[ProducesResponseType((int)HttpStatusCode.Forbidden)]
[ProducesResponseType((int)HttpStatusCode.InternalServerError)]
public ActionResult<HistoriaDTO> Get(int id)
{
    try
    {
        return Ok(_obtenerHistoriaPorIdCu.Ejecutar(id));

    catch (HistoriaException hex)
    {
        return BadRequest(hex.Message);
    }
    catch (Exception ex)
    {
        return StatusCode(500, ex.Message);
    }
}
```

## 3. Implementación de la consulta

Se reutiliza `IRepositorioHistorias.FindById`, que ya carga los capítulos de la historia.
```csharp
using Dominio.Entidades;
using Dominio.Excepciones;
using Dominio.InterfaceDeRepositorio;
using DTOs.DTOs;
using LogicaAplicacion.InterfacesDeCasoDeUso.Historias;

namespace LogicaAplicacion.CasosDeUso.Historias
{
    public class ObtenerFinalesPorHistoriaCU : IObtenerFinalesPorHistoria
    {
        private readonly IRepositorioHistorias repositorioHistorias;

        public ObtenerFinalesPorHistoriaCU(
            IRepositorioHistorias repositorioHistorias)
        {
            this.repositorioHistorias = repositorioHistorias;
        }

        public FinalesHistoriaDTO Ejecutar(int idHistoria)
        {
            if (idHistoria <= 0)
            {
                throw new HistoriaException(
                    "El identificador de la historia no es válido.");
            }

            Historia historia = repositorioHistorias.FindById(idHistoria);

            List<Final> finales = historia.Capitulos
                .OfType<Final>()
                .ToList();

            int mayorCantidad = finales.Any()
                ? finales.Max(final => final.CantLlegaron.Valor)
                : 0;

            return new FinalesHistoriaDTO
            {
                IdHistoria = historia.Id,
                TituloHistoria = historia.Titulo.Valor,

                Finales = finales
                    .Select(final => new FinalFrecuenciaDTO
                    {
                        IdFinal = final.Id,
                        Titulo = final.Titulo.Valor,
                        CantidadVecesAlcanzado =
                            final.CantLlegaron.Valor,

                        // Si hay empate, todos los finales con el
                        // valor máximo se consideran más frecuentes.
                        EsMasFrecuente =
                            mayorCantidad > 0 &&
                            final.CantLlegaron.Valor == mayorCantidad
                    })
                    .OrderByDescending(final =>
                        final.CantidadVecesAlcanzado)
                    .ThenBy(final => final.Titulo)
                    .ToList()
            };
        }
    }
}
```

La cantidad se obtiene desde `Final.CantLlegaron`, que ya es incrementada en el caso de uso existente cuando la lectura llega a un final:
```csharp
if (destino is Final final)
{
    lectura.RegistrarFinal(final);
    _repositorioLecturas.Update(lectura);

    final.RegistrarLlegada();
    _repositorioFinales.Update(final);
}
```

Por lo tanto, no es necesario volver a contar las lecturas en el nuevo caso de uso.

## 7. Consumir el endpoint desde `ProyectoSecundario`

La acción del controlador no contiene la consulta ni calcula el máximo. Solo realiza el consumo HTTP:
```csharp
private string baseUrlFinales =
    "http://localhost:5281/api/Historia";
```
```csharp
[HttpGet]
public IActionResult Finales(int idHistoria)
{
    string? token = HttpContext.Session.GetString("token");
    string? rol = HttpContext.Session.GetString("Rol");

    if (string.IsNullOrEmpty(token))
    {
        return RedirectToAction("Login", "Auth");
    }

    if (rol != "Administrador")
    {
        return RedirectToAction(nameof(Index));
    }

    if (idHistoria <= 0)
    {
        ViewBag.Error = "Debe seleccionar una historia.";
        return View();
    }

    HttpResponseMessage respuesta =
        ClienteHttpAuxiliar.EnviarSolicitud(
            baseUrlFinales + "/" + idHistoria + "/Finales",
            VerbosHttp.GET,
            token: token
        );

    if (!respuesta.IsSuccessStatusCode)
    {
        ViewBag.Error =
            "No se pudo obtener la información de los finales.";

        return View();
    }

    FinalesHistoriaModel? reporte =
        JsonConvert.DeserializeObject<FinalesHistoriaModel>(
            ClienteHttpAuxiliar.ObtenerBody(respuesta));

    if (reporte == null)
    {
        ViewBag.Error = "La respuesta del servidor no es válida.";
        return View();
    }

    return View(reporte);
}
```

De esta forma:

- `ProyectoObligatorio` accede al repositorio.
- `ProyectoObligatorio` aplica la lógica del requerimiento.
- `ProyectoObligatorio` determina el final más frecuente.
- `ProyectoSecundario` solamente consume y presenta el resultado.
- El endpoint queda protegido con el rol `Administrador`.

namespace LogicaAplicacion.InterfacesDeCasoDeUso.Lecturas

```csharp
using Dominio.Entidades;
using Dominio.InterfaceDeRepositorio;
using DTOs.DTOs;
using LogicaAplicacion.InterfacesDeCasoDeUso.Lecturas;

namespace LogicaAplicacion.CasosDeUso.Lecturas
{
    public class ObtenerRankingUsuariosCU : IObtenerRankingUsuarios
    {
        private IRepositorioLecturas repositorio;

        public ObtenerRankingUsuariosCU(IRepositorioLecturas repositorioCu)
        {
            repositorio = repositorioCu;
        }

        public List<UsuarioRankingDTO> Ejecutar()
        {
            List<Lectura> lecturasCompletadas =
                repositorio.FindAllCompletadas();

            return lecturasCompletadas
                .Where(lectura =>
                    lectura.Usuario != null &&
                    lectura.idUsuario.HasValue &&
                    lectura.idHistoria.HasValue)
                .GroupBy(lectura => new
                {
                    IdUsuario = lectura.idUsuario!.Value,
                    NombreUsuario = lectura.Usuario.NombreUsuario
                })
                .Select(grupo => new UsuarioRankingDTO
                {
                    IdUsuario = grupo.Key.IdUsuario,
                    NombreUsuario = grupo.Key.NombreUsuario,

                    // Una historia cuenta una sola vez por usuario.
                    CantidadHistoriasCompletadas = grupo
                        .Select(lectura => lectura.idHistoria!.Value)
                        .Distinct()
                        .Count()
                })
                .OrderByDescending(usuario =>
                    usuario.CantidadHistoriasCompletadas)
                .ThenBy(usuario => usuario.NombreUsuario)
                .ToList();
        }
    }
}
```

## 7. Consumo desde `HistoriaController`
```csharp
private string baseUrlRankingUsuarios =
    "http://localhost:5281/api/Lectura/RankingUsuarios";

public IActionResult RankingUsuarios()
{
    string? token = HttpContext.Session.GetString("token");
    string? rol = HttpContext.Session.GetString("Rol");

    if (string.IsNullOrEmpty(token))
    {
        return RedirectToAction("Login", "Auth");
    }

    if (rol != "Administrador")
    {
        return RedirectToAction(nameof(Index));
    }

    HttpResponseMessage respuesta =
        ClienteHttpAuxiliar.EnviarSolicitud(
            baseUrlRankingUsuarios,
            VerbosHttp.GET,
            token: token
        );

    List<UsuarioRankingModel> ranking =
        new List<UsuarioRankingModel>();

    if (respuesta.IsSuccessStatusCode)
    {
        ranking =
            JsonConvert.DeserializeObject<List<UsuarioRankingModel>>(
                ClienteHttpAuxiliar.ObtenerBody(respuesta))
            ?? new List<UsuarioRankingModel>();
    }
    else
    {
        ViewBag.Error =
            "No se pudo obtener el ranking de usuarios.";
    }

    return View(ranking);
}
```
